  ❌ UI → ChatService → Manual API → Cache Chaos

After (NEW):
  final messagesAsync = ref.watch(messagesStreamProvider(conversationId));
  messagesAsync.when(
    data: (messages) => ListView(...),
    loading: () => Spinner(),
    error: (e, s) => ErrorWidget(),
  );
